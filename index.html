Q1. Write a program that takes a positive integer from user & display the following in your browser: a. number, b. round off value, c. floor value, d. ceil value.

CODE:

var positiveNumber = parseFloat(prompt("Enter a positive floating-point number:"));
if (positiveNumber > 0) {
  var result = "number: " + positiveNumber + 
               "\nround off value: " + Math.round(positiveNumber) + 
               "\nfloor value: " + Math.floor(positiveNumber) + 
               "\nceil value: " + Math.ceil(positiveNumber);
  displayOutput("output1", result);
}

Q3. Write a program that displays the absolute value of a number. E.g. absolute value of -4 is 4 & absolute value of 5 is 5.

CODE:

var num = parseFloat(prompt("Enter any number to get its absolute value:"));
var absoluteValue = Math.abs(num);
var result = "The absolute value of " + num + " is " + absoluteValue;
displayOutput("output3", result);

Q4. Write a program that simulates a dice using random() method of JS Math class. Display the value of dice in your browser.

CODE:

var diceValue = Math.floor(Math.random() * 6) + 1;
var result = "random dice value: " + diceValue;
displayOutput("output4", result);

Q5. Write a program that simulates a coin toss using random() method of JS Math class. Display the value of coin in your browser.

CODE:

var toss = Math.floor(Math.random() * 2) + 1;
var coinValue = "";
if (toss === 2) {
  coinValue = "Heads";
} else {
  coinValue = "Tails";
}
var result = "random coin value: " + coinValue;
displayOutput("output5", result);

Q6. Write a program that shows a random number between 1 and 100 in your browser.

displayOutput("output6", result);

Helper function to show output on screen

CODE:

function displayOutput(id, text) {
  var element = document.getElementById(id);
  element.style.display = "block";
  element.innerText = text;
}
